#     pip install -r requirements.txt -r requirements-dev.txt

pytest>=8,<10

# psutil  — because the suite tests ONE app module that is pure logic and has no

# because the alternative is what happened: the singleton was cut during a move
# and the first thing to notice was CI, on three runners, after a push.
psutil>=5.9,<8
